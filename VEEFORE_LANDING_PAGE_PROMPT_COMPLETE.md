# ══════════════════════════════════════════════════════════════
# ⚡ SCROLL PATH SYSTEM — READ THIS COMPLETELY BEFORE CODING
# ══════════════════════════════════════════════════════════════

This is the single most important visual feature of the page. A glowing SVG bezier path runs the entire height of the page, drawing itself progressively as the user scrolls — like a neon wire that guides the eye through every section of the story.

## What It Does

- **Draws on scroll** — `stroke-dashoffset` animates from full length → 0 as user scrolls top → bottom
- **Winds organically** — path visits left / right / center in an S-curve, NOT a straight line
- **Gradient colour** — coral → gold → cyan → mint → rose → cyan → coral (no purple ever)
- **Dual-layer glow** — thin 1.5px crisp line + wide 8px blurred copy = neon glow effect
- **Moving tip dot** — a bright 4px circle rides the frontier, colour-shifting to match gradient zone
- **Section nodes** — at each section a dot springs in + fires an expanding ripple ring + pulses forever
- **Section labels** — tiny 10px JetBrains Mono labels appear beside each node

## Z-Index Rules (Critical)

```
z-index: 2  → Scroll path SVG  (above section backgrounds, BELOW all content)
z-index: 3+ → All page content (nav, text, cards, buttons)
```

- `pointer-events: none` on the SVG container — path NEVER intercepts clicks
- Hidden on mobile (`max-width: 768px`) — performance
- Hidden when `prefers-reduced-motion` is active — accessibility

## STEP 4 — Create `js/scroll-path.js`

Copy this file exactly. All configuration values are in the `config` object at the top.

```javascript
/* ════════════════════════════════════════════════════════════════
   VeefScrollPath
   Animated SVG bezier path that draws on scroll.
   Connects all landing page sections with glowing nodes.

   HOW IT WORKS:
   1. Injects an <svg> into <body> spanning the full page height
   2. Reads each section's offsetTop to calculate waypoints
   3. Generates a smooth cubic bezier through all waypoints
   4. Uses stroke-dashoffset to "draw" the path as user scrolls
   5. Moves a glowing tip dot along the frontier point
   6. Activates section nodes (spring pop + ripple) as path passes

   DEPENDENCIES:
   - GSAP + ScrollTrigger (preferred — smoothest result)
   - Falls back to native rAF scroll listener if GSAP unavailable

   USAGE: Just include the file. Auto-boots on DOMContentLoaded.
   ════════════════════════════════════════════════════════════════ */

class VeefScrollPath {

  /* ── Configuration ─────────────────────────────────────────────
     Edit these values to tune look and behaviour.
     ─────────────────────────────────────────────────────────────── */
  config = {

    /* X-positions the path visits per section (fraction of page width).
       Pattern repeats if sections outnumber entries.
       0.0 = left edge | 0.5 = center | 1.0 = right edge          */
    xPattern: [
      0.50, // hero         — center
      0.15, // problem      — far left
      0.82, // features     — far right
      0.50, // how-it-works — center
      0.25, // demo         — center-left
      0.75, // pricing      — center-right
      0.18, // testimonials — far left
      0.80, // faq          — far right
      0.50, // cta          — center
    ],

    /* Section IDs — must match HTML exactly */
    sectionIds: [
      '#hero', '#problem', '#features', '#how-it-works',
      '#demo', '#pricing', '#testimonials', '#faq', '#cta',
    ],

    /* Short labels shown beside each section node */
    nodeLabels: [
      'Hero', 'Problem', 'Features', 'How It Works',
      'Live Demo', 'Pricing', 'Creators Say', 'FAQ', 'Get Started',
    ],

    /* Gradient colour stops top → bottom — NO purple */
    gradientStops: [
      { offset: '0%',   color: '#FF4D2E' }, // coral
      { offset: '18%',  color: '#FFB800' }, // gold
      { offset: '36%',  color: '#00D4FF' }, // cyan
      { offset: '54%',  color: '#00FF87' }, // mint
      { offset: '72%',  color: '#FF2D7A' }, // rose
      { offset: '90%',  color: '#00D4FF' }, // cyan
      { offset: '100%', color: '#FF4D2E' }, // coral
    ],

    /* Dot colours per section node (matches sectionIds order) */
    nodeColors: [
      '#FF4D2E', '#FFB800', '#00D4FF',
      '#00FF87', '#FF2D7A', '#FFB800',
      '#00D4FF', '#FF4D2E', '#00FF87',
    ],

    tension:         0.48,  // bezier looseness: 0 = sharp corners, 1 = very loose
    mainStrokeWidth: 1.5,   // thin crisp line width
    glowStrokeWidth: 8,     // wide blurred copy width (creates glow)
    glowOpacity:     0.22,  // opacity of blurred copy
    tipDotRadius:    4,     // frontier dot size (px)
    nodeDotRadius:   3.5,   // section node dot size (px)
    scrub:           0.6,   // GSAP scrub lag (higher = smoother but laggier)
  };

  /* ── Internal state — do not edit ── */
  svg        = null;
  mainPath   = null;
  glowPath   = null;
  tipDot     = null;
  gradient   = null;
  waypoints  = [];
  nodes      = [];
  pathLength = 0;

  /* ── Constructor ── */
  constructor() {
    // Skip on mobile — performance
    if (window.matchMedia('(max-width: 768px)').matches) return;
    // Skip on reduced motion — accessibility
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;

    this._buildSVG();
    this._recalculate();
    this._bindScroll();

    // Recalculate if fonts or images change the page height
    window.addEventListener('load',   () => this._recalculate());
    window.addEventListener('resize', this._debounce(() => this._recalculate(), 300));
  }

  /* ── Build the SVG skeleton (called once) ── */
  _buildSVG() {
    const wrap = document.createElement('div');
    wrap.className = 'scroll-path-container';

    this.svg = this._ns('svg');
    this.svg.id = 'scroll-path-svg';

    /* ─ Defs ─ */
    const defs = this._ns('defs');

    /* Linear gradient — runs top to bottom of the full page.
       y2 is updated in _recalculate() to match real page height. */
    this.gradient = this._ns('linearGradient');
    this.gradient.id = 'veefPathGrad';
    this.gradient.setAttribute('gradientUnits', 'userSpaceOnUse');
    this.gradient.setAttribute('x1', '0'); this.gradient.setAttribute('x2', '0');
    this.gradient.setAttribute('y1', '0'); this.gradient.setAttribute('y2', '1000');

    this.config.gradientStops.forEach(s => {
      const stop = this._ns('stop');
      stop.setAttribute('offset',     s.offset);
      stop.setAttribute('stop-color', s.color);
      this.gradient.appendChild(stop);
    });

    defs.append(
      this.gradient,
      this._makeFilter('veefPathGlow', 5),  // path glow blur
      this._makeFilter('veefTipGlow',  8),  // tip dot glow blur
      this._makeFilter('veefNodeGlow', 6),  // section node glow blur
    );
    this.svg.appendChild(defs);

    /* ─ Glow path — wide + blurred copy for the neon effect ─ */
    this.glowPath = this._ns('path');
    this.glowPath.id = 'scroll-glow-path';
    this.glowPath.setAttribute('stroke',         'url(#veefPathGrad)');
    this.glowPath.setAttribute('stroke-width',   this.config.glowStrokeWidth);
    this.glowPath.setAttribute('fill',           'none');
    this.glowPath.setAttribute('opacity',        this.config.glowOpacity);
    this.glowPath.setAttribute('filter',         'url(#veefPathGlow)');
    this.glowPath.setAttribute('stroke-linecap', 'round');

    /* ─ Main path — thin, sharp, visible line ─ */
    this.mainPath = this._ns('path');
    this.mainPath.id = 'scroll-main-path';
    this.mainPath.setAttribute('stroke',         'url(#veefPathGrad)');
    this.mainPath.setAttribute('stroke-width',   this.config.mainStrokeWidth);
    this.mainPath.setAttribute('fill',           'none');
    this.mainPath.setAttribute('stroke-linecap', 'round');

    /* ─ Tip dot — moves at the drawn frontier ─ */
    this.tipDot = this._ns('circle');
    this.tipDot.className.baseVal = 'path-tip-dot';
    this.tipDot.setAttribute('r',       this.config.tipDotRadius);
    this.tipDot.setAttribute('fill',    '#FFFFFF');
    this.tipDot.setAttribute('filter',  'url(#veefTipGlow)');
    this.tipDot.setAttribute('opacity', '0');

    this.svg.append(this.glowPath, this.mainPath, this.tipDot);
    wrap.appendChild(this.svg);

    // Prepend to body — sits below all content in stacking context
    document.body.prepend(wrap);
  }

  /* ── Recalculate geometry on load / resize ── */
  _recalculate() {
    const W = document.documentElement.clientWidth;
    const H = document.documentElement.scrollHeight;

    this.svg.setAttribute('width',  W);
    this.svg.setAttribute('height', H);
    this.svg.style.height = H + 'px';

    // Stretch gradient to real page height
    this.gradient.setAttribute('y2', H);

    /* ─ Build waypoints ─ */
    this.waypoints = [];

    // Entry: top-center of page
    this.waypoints.push({ x: W * 0.5, y: 60 });

    // One waypoint per section — reads from live DOM
    this.config.sectionIds.forEach((id, i) => {
      const el = document.querySelector(id);
      if (!el) return;
      const rect = el.getBoundingClientRect();
      const y    = rect.top + window.scrollY + rect.height * 0.45;
      const x    = W * this.config.xPattern[i % this.config.xPattern.length];
      this.waypoints.push({
        x, y,
        label: this.config.nodeLabels[i],
        color: this.config.nodeColors[i],
      });
    });

    // Exit: bottom-center of page
    this.waypoints.push({ x: W * 0.5, y: H - 60 });

    /* ─ Build path string and apply ─ */
    const d = this._buildBezier(this.waypoints);
    this.mainPath.setAttribute('d', d);
    this.glowPath.setAttribute('d', d);

    /* ─ Dash setup — start fully undrawn ─ */
    this.pathLength = this.mainPath.getTotalLength();
    [this.mainPath, this.glowPath].forEach(p => {
      p.style.strokeDasharray  = this.pathLength;
      p.style.strokeDashoffset = this.pathLength; // fully hidden at start
    });

    /* ─ Rebuild section nodes ─ */
    this._buildNodes();
  }

  /* ── Generate smooth cubic bezier path string through waypoints ── */
  _buildBezier(pts) {
    if (pts.length < 2) return '';
    let d = `M ${R(pts[0].x)} ${R(pts[0].y)}`;
    for (let i = 1; i < pts.length; i++) {
      const a   = pts[i - 1];
      const b   = pts[i];
      const dy  = b.y - a.y;
      const ten = this.config.tension;
      // Cubic bezier: control points pull handles vertically for organic S-curve
      d += ` C ${R(a.x)} ${R(a.y + dy * ten)}, ${R(b.x)} ${R(b.y - dy * ten)}, ${R(b.x)} ${R(b.y)}`;
    }
    return d;
  }

  /* ── Create glowing node dots at each section waypoint ── */
  _buildNodes() {
    // Remove existing nodes
    this.nodes.forEach(n => { n.dot?.remove(); n.label?.remove(); });
    this.nodes = [];

    const W = document.documentElement.clientWidth;

    // Skip first (entry) and last (exit) waypoints — only section waypoints
    const sectionWPs = this.waypoints.slice(1, -1);

    sectionWPs.forEach(wp => {
      const color   = wp.color || '#FF4D2E';
      const isRight = wp.x > W * 0.5; // determines label side

      /* ─ Node dot ─ */
      const dot = this._ns('circle');
      dot.setAttribute('cx',      R(wp.x));
      dot.setAttribute('cy',      R(wp.y));
      dot.setAttribute('r',       this.config.nodeDotRadius);
      dot.setAttribute('fill',    color);
      dot.setAttribute('opacity', '0'); // hidden until activated
      dot.setAttribute('filter',  'url(#veefNodeGlow)');
      dot.className.baseVal = 'path-node-inner';

      /* ─ Section label — opposite side from path direction ─ */
      const label = this._ns('text');
      label.setAttribute('x',           R(isRight ? wp.x - 16 : wp.x + 16));
      label.setAttribute('y',           R(wp.y + 4));
      label.setAttribute('text-anchor', isRight ? 'end' : 'start');
      label.className.baseVal = 'path-section-label';
      label.textContent = wp.label || '';

      this.svg.appendChild(dot);
      this.svg.appendChild(label);
      this.nodes.push({ dot, label, y: wp.y, color, activated: false });
    });
  }

  /* ── Bind scroll events to the path drawing ── */
  _bindScroll() {
    if (typeof gsap !== 'undefined' && typeof ScrollTrigger !== 'undefined') {
      /* Preferred: GSAP ScrollTrigger — buttery smooth with scrub */
      gsap.to({}, {
        scrollTrigger: {
          trigger: 'body',
          start:   'top top',
          end:     'bottom bottom',
          scrub:   this.config.scrub,
          onUpdate: self => this._onProgress(self.progress),
        },
      });
    } else {
      /* Fallback: native rAF-throttled scroll listener */
      let pending = false;
      window.addEventListener('scroll', () => {
        if (pending) return;
        pending = true;
        requestAnimationFrame(() => {
          const prog = Math.min(
            window.scrollY / (document.documentElement.scrollHeight - window.innerHeight),
            1
          );
          this._onProgress(prog);
          pending = false;
        });
      }, { passive: true });
    }
  }

  /* ── Called every frame with scroll progress 0 → 1 ── */
  _onProgress(progress) {
    const remaining = this.pathLength * (1 - progress);

    // Draw path by reducing dashoffset
    this.mainPath.style.strokeDashoffset = remaining;
    this.glowPath.style.strokeDashoffset = remaining;

    // Move tip dot to current frontier point
    if (progress > 0.005 && this.pathLength > 0) {
      try {
        const pt = this.mainPath.getPointAtLength(this.pathLength * progress);
        this.tipDot.setAttribute('cx',      R(pt.x));
        this.tipDot.setAttribute('cy',      R(pt.y));
        this.tipDot.setAttribute('opacity', '1');

        // Match tip dot colour to current gradient zone
        const stops = this.config.gradientStops;
        const idx   = Math.min(Math.floor(progress * (stops.length - 1)), stops.length - 2);
        this.tipDot.setAttribute('fill', stops[idx].color);
      } catch (_) { /* path not ready yet — safe to ignore */ }
    } else {
      this.tipDot.setAttribute('opacity', '0');
    }

    this._checkNodes(progress);
  }

  /* ── Activate section node when path frontier passes its Y position ── */
  _checkNodes(progress) {
    if (!this.pathLength) return;
    let drawnY = 0;
    try {
      drawnY = this.mainPath.getPointAtLength(this.pathLength * progress).y;
    } catch (_) { return; }

    this.nodes.forEach(node => {
      if (!node.activated && node.y <= drawnY + 20) {
        node.activated = true;
        this._activateNode(node);
      }
    });
  }

  /* ── Spring pop-in + ripple ring + idle pulse when node activates ── */
  _activateNode(node) {
    node.dot.setAttribute('opacity', '1');
    node.label.classList.add('visible');

    if (typeof gsap !== 'undefined') {
      // Spring pop-in animation
      gsap.fromTo(node.dot,
        { attr: { r: 0 } },
        { attr: { r: this.config.nodeDotRadius }, duration: 0.35, ease: 'back.out(3)' }
      );

      // Expanding ripple ring
      const ripple = this._ns('circle');
      ripple.setAttribute('cx',           node.dot.getAttribute('cx'));
      ripple.setAttribute('cy',           node.dot.getAttribute('cy'));
      ripple.setAttribute('r',            this.config.nodeDotRadius);
      ripple.setAttribute('fill',         'none');
      ripple.setAttribute('stroke',       node.color);
      ripple.setAttribute('stroke-width', '1.2');
      ripple.setAttribute('opacity',      '0.8');
      this.svg.insertBefore(ripple, node.dot);

      gsap.to(ripple, {
        attr:       { r: 22, opacity: 0 },
        duration:   0.9,
        ease:       'power2.out',
        onComplete: () => ripple.remove(),
      });

      // Gentle forever pulse on the dot
      gsap.to(node.dot, {
        attr:     { r: this.config.nodeDotRadius * 1.6 },
        repeat:   -1,
        yoyo:     true,
        duration: 1.4,
        ease:     'sine.inOut',
        delay:    0.5,
      });
    }
    // No-GSAP fallback: dot is simply shown (opacity already set above)
  }

  /* ── SVG namespace helper ── */
  _ns(tag) {
    return document.createElementNS('http://www.w3.org/2000/svg', tag);
  }

  /* ── Build an SVG feGaussianBlur glow filter ── */
  _makeFilter(id, blurAmount) {
    const f  = this._ns('filter');
    f.id = id;
    f.setAttribute('x',      '-100%');
    f.setAttribute('y',      '-100%');
    f.setAttribute('width',  '300%');
    f.setAttribute('height', '300%');
    const g  = this._ns('feGaussianBlur');
    g.setAttribute('stdDeviation', blurAmount);
    g.setAttribute('result',       'b');
    const m  = this._ns('feMerge');
    const n1 = this._ns('feMergeNode'); n1.setAttribute('in', 'b');
    const n2 = this._ns('feMergeNode'); n2.setAttribute('in', 'SourceGraphic');
    m.append(n1, n2);
    f.append(g, m);
    return f;
  }

  /* ── Debounce helper ── */
  _debounce(fn, ms) {
    let t;
    return (...args) => { clearTimeout(t); t = setTimeout(() => fn(...args), ms); };
  }
}

/* Round to 2 decimal places — keeps SVG path data clean */
function R(n) { return Math.round(n * 100) / 100; }

/* ── Boot — slight delay so all section heights are finalised ── */
document.addEventListener('DOMContentLoaded', () => {
  setTimeout(() => { window.veefPath = new VeefScrollPath(); }, 180);
});
```
